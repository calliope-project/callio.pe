{{/* Renders one publication as a citation. Params: dict "entry" "compact" Compact mode drops the venue and links, for use inside cards. */}} {{- $e := .entry -}} {{- $authors := "" -}} {{- with $e.authors_display -}} {{- $authors = . -}} {{- else -}} {{- $n := len $e.authors -}} {{- if eq $n 1 -}} {{- $authors = index $e.authors 0 -}} {{- else if le $n 4 -}} {{- $authors = delimit $e.authors ", " " and " -}} {{- else -}} {{- $authors = printf "%s et al." (index $e.authors 0) -}} {{- end -}} {{- end -}} {{- $link := "" -}} {{- with $e.doi }}{{ $link = printf "https://doi.org/%s" . }}{{ else }}{{ $link = $e.url }}{{ end -}} {{- if .compact -}} {{ $authors }} ({{ $e.year }}) {{- else -}} {{ $authors }} ({{ $e.year }}). {{ $e.title }}. {{- with $e.venue_detail }} {{ . }}.{{ else }}{{ with $e.venue }} {{ . }}.{{ end }}{{ end }} {{- with $e.doi }} doi: {{ . }}{{ end }} {{- if and (not $e.doi) $e.url }} Full text{{ end }} {{- if $e.open_access }} OA{{ end }} {{- end -}}
